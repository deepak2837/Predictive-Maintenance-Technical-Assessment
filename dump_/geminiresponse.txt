Write a Python function that takes an array of real numbers and returns the maximum drawdown. Include a test case. Explain the time and space complexity of your solution above.Pythondef calculate_max_drawdown(prices):
    if not prices or len(prices) < 2:
        return 0.0
    
    max_price = prices[0]
    max_drawdown = 0.0
    
    for price in prices:
        if price > max_price:
            max_price = price
        
        drawdown = (max_price - price) / max_price
        if drawdown > max_drawdown:
            max_drawdown = drawdown
            
    return max_drawdown

import yfinance as yf
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, classification_report
from sklearn.model_selection import train_test_split
import matplotlib.pyplot as plt

# 1. Data Collection
# We will pull 10 years of SPY (S&P 500 ETF) daily data.
ticker = "SPY"
data = yf.download(ticker, start="2014-01-01", end="2024-01-01")

# 2. Data Exploration and Preprocessing
# Calculate daily returns
data['Return'] = data['Close'].pct_change()

# Target Variable: 1 if return > 0 (Up day), 0 otherwise (Down day)
data['Target'] = np.where(data['Return'].shift(-1) > 0, 1, 0)

# Feature Engineering
# Simple Moving Averages
data['SMA_10'] = data['Close'].rolling(window=10).mean()
data['SMA_50'] = data['Close'].rolling(window=50).mean()

# Rolling Volatility (10-day)
data['Vol_10'] = data['Return'].rolling(window=10).std()

# Lagged Returns
data['Lag_1'] = data['Return'].shift(1)
data['Lag_2'] = data['Return'].shift(2)

# Drop missing values resulting from rolling windows and lags
data.dropna(inplace=True)

# Define features matrix (X) and target vector (y)
features = ['SMA_10', 'SMA_50', 'Vol_10', 'Lag_1', 'Lag_2']
X = data[features]
y = data['Target']

# 3. Model Development
# Chronological train-test split (80% train, 20% test) to prevent data leakage in time series
split_idx = int(len(data) * 0.8)
X_train, X_test = X.iloc[:split_idx], X.iloc[split_idx:]
y_train, y_test = y.iloc[:split_idx], y.iloc[split_idx:]

# Initialize and train Random Forest Classifier
model = RandomForestClassifier(n_estimators=100, max_depth=5, random_state=42)
model.fit(X_train, y_train)

# 4. Results and Interpretation
# Predictions
predictions = model.predict(X_test)

# Accuracy
accuracy = accuracy_score(y_test, predictions)
print(f"Model Accuracy: {accuracy:.4f}")
print("\nClassification Report:\n", classification_report(y_test, predictions))

# Feature Importance
importances = model.feature_importances_
indices = np.argsort(importances)[::-1]

print("\nFeature Ranking:")
for f in range(X.shape[1]):
    print(f"{f + 1}. {features[indices[f]]} ({importances[indices[f]]:.4f})")
